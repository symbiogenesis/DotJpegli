global using DotJpegli.Compatibility;
using DotJpegli;
using DotJpegli.Internal;

foreach (int width in new[] { 1, 19, 65 })
{
    const int height = 35;
    byte[] pixels = new byte[width * height * 3];
    for (int index = 0; index < pixels.Length; index++)
    {
        pixels[index] = (byte)((index * 1237) ^ (index / 7));
    }
    PixelSource source = new(pixels, PixelLayout.Rgb24);
    foreach (int sampling in new[] { 1, 2 })
    {
        using var baseline = new BaselineAdaptiveQuantization(width, height, sampling);
        using var stencil = new StencilAdaptiveQuantization(width, height, sampling);
        using var current = new AdaptiveQuantization(width, height, sampling);
        float[] expected = new float[baseline.BlocksWide * sampling];
        float[] actual = new float[expected.Length];
        float[] stencilActual = new float[expected.Length];
        for (int path = 0; path < 3; path++)
        {
            if (path == 0)
            {
                baseline.ComputeRowScalar(source, width * 3, 0, 18, expected, default);
                stencil.ComputeRowScalar(source, width * 3, 0, 18, stencilActual, default);
                current.ComputeRowScalar(source, width * 3, 0, 18, actual, default);
            }
            else if (path == 1)
            {
                baseline.ComputeRowVector(source, width * 3, 0, 18, expected, default);
                stencil.ComputeRowVector(source, width * 3, 0, 18, stencilActual, default);
                current.ComputeRowVector(source, width * 3, 0, 18, actual, default);
            }
            else
            {
                baseline.ComputeRowBatched(source, width * 3, 0, 18, expected, default);
                stencil.ComputeRowBatched(source, width * 3, 0, 18, stencilActual, default);
                current.ComputeRowBatched(source, width * 3, 0, 18, actual, default);
            }
            if (!expected.Select(BitConverter.SingleToInt32Bits).SequenceEqual(actual.Select(BitConverter.SingleToInt32Bits)))
            {
                throw new InvalidOperationException("AQ bits changed between revisions.");
            }
            if (!expected.Select(BitConverter.SingleToInt32Bits).SequenceEqual(stencilActual.Select(BitConverter.SingleToInt32Bits)))
            {
                throw new InvalidOperationException("AQ bits changed after stencil revision.");
            }
        }
    }
}
Console.WriteLine("Untimed baseline/current AQ field bits match; code-generation probe completed.");
