namespace DotJpegli.Internal;

using System.Numerics;
using System.Runtime.CompilerServices;
#if !NETSTANDARD2_1
using System.Runtime.Intrinsics;
#endif

// Value-type specialization keeps research calls and their register spills out
// of ordinary AQ loops. Instance interface members also work on .NET Standard.
internal interface IAqRationalMath
{
    public float Divide(float numerator, float denominator, AqRationalSite site);
    public Vector<float> Divide(Vector<float> numerator, Vector<float> denominator, AqRationalSite site);
#if !NETSTANDARD2_1
    public Vector256<float> Divide(Vector256<float> numerator, Vector256<float> denominator, AqRationalSite site);
#endif
}

internal readonly struct OriginalAqRationalMath : IAqRationalMath
{
    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    public float Divide(float numerator, float denominator, AqRationalSite site) => numerator / denominator;

    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    public Vector<float> Divide(Vector<float> numerator, Vector<float> denominator, AqRationalSite site) => numerator / denominator;

#if !NETSTANDARD2_1
    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    public Vector256<float> Divide(Vector256<float> numerator, Vector256<float> denominator, AqRationalSite site) => numerator / denominator;
#endif
}

internal readonly struct ReciprocalAqRationalMath(AqRationalDiagnostics? diagnostics) : IAqRationalMath
{
    private readonly AqRationalDiagnostics? diagnostics = diagnostics;

    public float Divide(float numerator, float denominator, AqRationalSite site)
    {
        float result = AqReciprocal.Divide(numerator, denominator, out bool fallback);
        this.diagnostics?.Record(site, fallback);
        return result;
    }

    public Vector<float> Divide(Vector<float> numerator, Vector<float> denominator, AqRationalSite site)
    {
        // Explicit research only: certify each lane and preserve the original
        // vector division's exceptional-value bits on every fallback lane.
        Span<float> results = stackalloc float[Vector<float>.Count];
        Span<bool> fallbacks = stackalloc bool[Vector<float>.Count];
        bool anyFallback = false;
        for (int lane = 0; lane < results.Length; lane++)
        {
            results[lane] = AqReciprocal.Divide(numerator[lane], denominator[lane], out bool fallback);
            fallbacks[lane] = fallback;
            anyFallback |= fallback;
            this.diagnostics?.Record(site, fallback);
        }
        if (anyFallback)
        {
            Vector<float> original = numerator / denominator;
            for (int lane = 0; lane < results.Length; lane++)
            {
                if (fallbacks[lane])
                {
                    results[lane] = original[lane];
                }
            }
        }
        this.diagnostics?.RecordVector(anyFallback, vector256: false);
        return new Vector<float>(results);
    }

#if !NETSTANDARD2_1
    public Vector256<float> Divide(Vector256<float> numerator, Vector256<float> denominator, AqRationalSite site)
    {
        Span<float> results = stackalloc float[8];
        Span<bool> fallbacks = stackalloc bool[8];
        bool anyFallback = false;
        for (int lane = 0; lane < results.Length; lane++)
        {
            results[lane] = AqReciprocal.Divide(numerator[lane], denominator[lane], out bool fallback);
            fallbacks[lane] = fallback;
            anyFallback |= fallback;
            this.diagnostics?.Record(site, fallback);
        }
        if (anyFallback)
        {
            Vector256<float> original = numerator / denominator;
            for (int lane = 0; lane < results.Length; lane++)
            {
                if (fallbacks[lane])
                {
                    results[lane] = original[lane];
                }
            }
        }
        this.diagnostics?.RecordVector(anyFallback, vector256: true);
        return Vector256.Create(results);
    }
#endif
}
