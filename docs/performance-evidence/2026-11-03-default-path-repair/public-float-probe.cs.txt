using System;
using System.Collections.Generic;
using System.IO;
using System.Security.Cryptography;
using System.Text.Json;
using DotJpegli;

internal static class Program
{
    private static void Main()
    {
        const int width = 65, height = 17;
        var exceptional = new float[width * height * 3];
        var normalized = new float[exceptional.Length];
        for (int index = 0; index < exceptional.Length; index++)
        {
            exceptional[index] = (index % 7) switch
            {
                0 => float.PositiveInfinity,
                1 => float.NegativeInfinity,
                2 => BitConverter.Int32BitsToSingle(0x7fc12345),
                3 => float.MaxValue,
                4 => -float.MaxValue,
                5 => -0f,
                _ => 0f
            };
            normalized[index] = (index % 7) switch
            {
                0 or 3 => 1f,
                2 => 0.5f,
                5 => -0f,
                _ => 0f
            };
        }
        var rows = new List<object>();
        foreach (bool progressive in new[] { false, true })
        {
            foreach (ChromaSubsampling sampling in new[] { ChromaSubsampling.None, ChromaSubsampling.Quarter })
            {
                var options = new JpegliEncoderOptions
                {
                    Distance = 1f,
                    AdaptiveQuantization = true,
                    OptimizeHuffmanTables = true,
                    ProgressiveLevel = progressive ? 2 : 0,
                    Subsampling = sampling,
                    MaxDegreeOfParallelism = 1
                };
                using var output = new MemoryStream();
                try
                {
                    JpegliEncoder.Encode(output, exceptional, width, height, options: options);
                    throw new InvalidOperationException("Exceptional public samples unexpectedly accepted.");
                }
                catch (ArgumentException exception)
                {
                    rows.Add(new
                    {
                        pattern = "exceptional-kind-3", progressive, sampling = (int)sampling,
                        exception = exception.GetType().FullName, exception.Message, exception.ParamName,
                        prefixLength = output.Length,
                        prefixSha256 = Convert.ToHexString(SHA256.HashData(output.ToArray()))
                    });
                }
                byte[] jpeg = JpegliEncoder.Encode(normalized, width, height, options: options);
                rows.Add(new
                {
                    pattern = "normalized-control", progressive, sampling = (int)sampling,
                    length = jpeg.Length, sha256 = Convert.ToHexString(SHA256.HashData(jpeg))
                });
            }
        }
        Console.WriteLine(JsonSerializer.Serialize(rows, new JsonSerializerOptions { WriteIndented = true }));
    }
}
