using DotJpegli;
using DotJpegli.Internal;

const int width = 65, height = 17;
float[] source = new float[width * height * 3];
for (int index = 0; index < source.Length; index++)
{
    source[index] = (index % 7) switch
    {
        0 => float.PositiveInfinity,
        1 => float.NegativeInfinity,
        2 => BitConverter.Int32BitsToSingle(0x7fc12345),
        3 => float.MaxValue,
        4 => -float.MaxValue,
        5 => -0f,
        _ => 0f
    };
}
PixelSource pixels = new(source, PixelComponentLayout.Rgb);
using var original = new AdaptiveQuantization(width, height, 2);
var diagnostics = new AqRationalDiagnostics();
using var research = new AdaptiveQuantization(width, height, 2,
    aqRationalKernel: KernelPreference.Optimized, aqRationalDiagnostics: diagnostics);
foreach (int top in new[] { 0, 16, 0 })
{
    for (int path = 0; path < 3; path++)
    {
        foreach (var variant in new[] { (Name: "Original", Value: original), (Name: "Research", Value: research) })
        {
            float[] values = new float[original.BlocksWide * 2];
            if (path == 0)
            {
                variant.Value.ComputeRowScalar(pixels, width * 3, top, 18, values, default);
            }
            else if (path == 1)
            {
                variant.Value.ComputeRowVector(pixels, width * 3, top, 18, values, default);
            }
            else
            {
                variant.Value.ComputeRowBatched(pixels, width * 3, top, 18, values, default);
            }
            Console.WriteLine($"{top}/{path}/{variant.Name}: {string.Join(',', values.Select(value => BitConverter.SingleToInt32Bits(value).ToString("X8")))}");
            if (top == 0 && path == 1 && variant.Name == "Original")
            {
                foreach (string field in new[] { "luma", "preErosion", "fuzzy" })
                {
                    float[] state = (float[])typeof(AdaptiveQuantization).GetField(field,
                        System.Reflection.BindingFlags.Instance | System.Reflection.BindingFlags.NonPublic)!.GetValue(variant.Value)!;
                    Console.WriteLine($"{field}: {string.Join(',', state.Select(value => BitConverter.SingleToInt32Bits(value).ToString("X8")))}");
                }
            }
        }
    }
}
