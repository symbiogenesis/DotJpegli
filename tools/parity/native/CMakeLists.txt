cmake_minimum_required(VERSION 3.16)
project(DotJpegliParityNative LANGUAGES C CXX)

set(JPEGLI_PIN "031a0077f5799a6041004267fc12b956c1f52a20")
set(JPEGLI_SOURCE_DIR "" CACHE PATH "Unmodified pinned google/jpegli Git checkout")
if(NOT IS_ABSOLUTE "${JPEGLI_SOURCE_DIR}")
  message(FATAL_ERROR "JPEGLI_SOURCE_DIR must be an absolute Git checkout path")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${JPEGLI_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE JPEGLI_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE GIT_RESULT)
if(NOT GIT_RESULT EQUAL 0 OR NOT JPEGLI_HEAD STREQUAL JPEGLI_PIN)
  message(FATAL_ERROR "Expected unmodified google/jpegli revision ${JPEGLI_PIN}; got ${JPEGLI_HEAD}")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${JPEGLI_SOURCE_DIR}" diff --quiet --ignore-submodules=none HEAD --
  RESULT_VARIABLE GIT_DIRTY)
if(NOT GIT_DIRTY EQUAL 0)
  message(FATAL_ERROR "Reference checkout has tracked changes")
endif()

# Build the original encoder/decoder, without tools, tests, or benchmarks.
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_DEVTOOLS OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_BENCHMARK OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_FUZZERS OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_JPEGLI_LIBJPEG OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_DOXYGEN OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_MANPAGES OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_JNI OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_SJPEG OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_OPENEXR OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(JPEGLI_ENABLE_SKCMS ON CACHE BOOL "" FORCE)
find_package(Threads REQUIRED)
add_subdirectory("${JPEGLI_SOURCE_DIR}" jpegli EXCLUDE_FROM_ALL)

add_executable(jpegli-parity-native main.cc)
target_compile_features(jpegli-parity-native PRIVATE cxx_std_17)
target_include_directories(jpegli-parity-native PRIVATE "${JPEGLI_SOURCE_DIR}")
target_compile_definitions(jpegli-parity-native PRIVATE JPEGLI_PIN="${JPEGLI_PIN}")
target_link_libraries(jpegli-parity-native PRIVATE jpegli-static)
